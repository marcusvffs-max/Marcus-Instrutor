-- Expose the WhatsApp assistant link only to student accounts linked to Marcus.
-- The WhatsApp integration is currently single-owner and uses the same fixed owner
-- configured by the webhook. Keep the lookup in the non-exposed private schema.
BEGIN;

CREATE OR REPLACE FUNCTION private.can_show_marcus_whatsapp_assistant_internal()
RETURNS boolean
LANGUAGE sql
STABLE
SECURITY DEFINER
SET search_path = ''
AS $function$
  SELECT
    auth.uid() IS NOT NULL
    AND EXISTS (
      SELECT 1
      FROM public.profiles p
      JOIN public.students s ON s.user_id = p.id
      WHERE p.id = auth.uid()
        AND p.role = 'usuario'
        AND s.owner_id = '8da26777-dad7-46dd-a332-4b5b80b027c7'::uuid
    );
$function$;

REVOKE ALL ON FUNCTION private.can_show_marcus_whatsapp_assistant_internal() FROM PUBLIC, anon;

LANGUAGE sql

SET search_path = ''
AS $function$
  SELECT private.can_show_marcus_whatsapp_assistant_internal();
$function$;
