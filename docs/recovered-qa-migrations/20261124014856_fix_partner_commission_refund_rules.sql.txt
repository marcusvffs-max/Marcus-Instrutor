create or replace function private.create_partner_commission_for_service()
returns trigger
language plpgsql
security definer
set search_path to ''
as $function$
declare
  v_sale public.sales%rowtype;
  v_partner public.partners%rowtype;
  v_base numeric(12,2);
  v_required integer;
  v_amount numeric(12,2);
  v_commission_rate numeric(5,2);
  v_lesson public.lessons%rowtype;
begin
  select * into v_sale from public.sales
   where id=new.sale_id and owner_id=new.owner_id;
  if not found or v_sale.student_id is null or v_sale.status<>'confirmada' then
    return new;
  end if;
  if coalesce(new.tipo_item,'servico')<>'servico' or new.fixed_item_key is not null then
    return new;
  end if;

  select p.* into v_partner
    from public.students s
    join public.partners p on p.id=s.partner_id and p.owner_id=s.owner_id
   where s.id=v_sale.student_id and s.owner_id=new.owner_id;
  if not found then return new; end if;

  select c.commission_percent into v_commission_rate
    from public.partner_commissions c
   where c.sale_item_id=new.id and c.owner_id=new.owner_id;
  if not found then
    if v_partner.status<>'aprovado' or v_partner.commission_percent<=0 then return new; end if;
    v_commission_rate := v_partner.commission_percent;
  end if;

  v_base := greatest(0,coalesce(new.valor_liquido,new.subtotal,0)-coalesce(new.valor_devolvido,0));
  v_required := greatest(0,coalesce(new.aulas_contratadas,0)-coalesce(new.aulas_devolvidas,0));
  v_amount := round(v_base*v_commission_rate/100,2);

  insert into public.partner_commissions(
    owner_id,partner_id,student_id,sale_id,sale_item_id,descricao,
    base_amount,commission_percent,commission_amount,lessons_required,
    status,available_at
  ) values (
    new.owner_id,v_partner.id,v_sale.student_id,v_sale.id,new.id,new.descricao,
    v_base,v_commission_rate,v_amount,v_required,
    case when v_base<=0 or v_required=0 then 'cancelado' else 'pendente' end,
    null
  )
  on conflict (sale_item_id) do update set
    base_amount=excluded.base_amount,
    commission_percent=partner_commissions.commission_percent,
    commission_amount=round(excluded.base_amount*partner_commissions.commission_percent/100,2),
    lessons_required=excluded.lessons_required,
    status=case
      when partner_commissions.status in ('pago','cancelado') then partner_commissions.status
      when excluded.status='cancelado' then 'cancelado'
      when partner_commissions.lessons_completed>=excluded.lessons_required then 'disponivel'
      else 'pendente'
    end,
    available_at=case
      when partner_commissions.status in ('pago','cancelado') then partner_commissions.available_at
      when excluded.status='cancelado' then null
      when partner_commissions.lessons_completed>=excluded.lessons_required
        then coalesce(partner_commissions.available_at,now())
      else null
    end
  where partner_commissions.status <> 'pago';

  if v_required > 0 and v_base > 0 then
    for v_lesson in
      select l.* from public.lessons l
       where l.owner_id=new.owner_id
         and l.student_id=v_sale.student_id
         and l.status='realizada'
         and l.data_aula >= v_sale.data_venda
         and not exists (
           select 1 from public.partner_commission_lessons cl
            where cl.lesson_id=l.id and cl.owner_id=l.owner_id
         )
       order by l.data_aula,l.horario,l.created_at,l.id
       limit v_required
    loop
      insert into public.partner_commission_lessons(owner_id,commission_id,lesson_id)
      select new.owner_id,c.id,v_lesson.id
        from public.partner_commissions c
       where c.sale_item_id=new.id and c.owner_id=new.owner_id
         and c.status not in ('cancelado','pago')
      on conflict (lesson_id) do nothing;
    end loop;
  end if;
  return new;
end;
$function$;

create or replace function private.guard_paid_partner_commission_refund()
returns trigger
language plpgsql
security definer
set search_path to ''
as $function$
begin
  if (coalesce(new.aulas_devolvidas,0) > coalesce(old.aulas_devolvidas,0)
      or coalesce(new.valor_devolvido,0) > coalesce(old.valor_devolvido,0))
     and exists (
       select 1 from public.partner_commissions c
        where c.sale_item_id=old.id and c.owner_id=old.owner_id and c.status='pago'
     ) then
    raise exception 'Não é possível devolver aulas: a comissão deste pacote já foi paga ao parceiro.'
      using errcode='23514';
  end if;
  return new;
end;
$function$;

revoke all on function private.guard_paid_partner_commission_refund() from public, anon, authenticated;
drop trigger if exists trg_guard_paid_partner_commission_refund on public.sale_items;
create trigger trg_guard_paid_partner_commission_refund
before update of aulas_devolvidas, valor_devolvido on public.sale_items
for each row execute function private.guard_paid_partner_commission_refund();

drop trigger if exists trg_create_partner_commission_for_service on public.sale_items;
create trigger trg_create_partner_commission_for_service
after insert or update of valor_liquido, valor_devolvido, aulas_devolvidas on public.sale_items
for each row execute function private.create_partner_commission_for_service();

create or replace function public.delete_my_sale(p_sale_id uuid)
returns jsonb
language plpgsql
set search_path to ''
as $function$
declare
  v_uid uuid := auth.uid();
  v_role text := private.current_app_role();
  v_sale public.sales%rowtype;
  v_student public.students%rowtype;
  v_net_lessons integer := 0;
begin
  if v_uid is null then
    raise exception 'Sessão não autenticada.' using errcode='42501';
  end if;
  if v_role is distinct from 'admin' and v_role is distinct from 'instrutor' then
    raise exception 'Acesso permitido somente ao administrador ou instrutor com autorização ativa.' using errcode='42501';
  end if;

  select * into v_sale from public.sales
   where id=p_sale_id and owner_id=v_uid for update;
  if not found then
    raise exception 'Venda não encontrada nesta conta.' using errcode='P0002';
  end if;

  if exists (
    select 1 from public.partner_commissions c
     where c.sale_id=p_sale_id and c.owner_id=v_uid
  ) then
    raise exception 'Esta venda possui comissão vinculada a um parceiro e não pode ser excluída. Preserve o histórico e utilize os procedimentos de devolução ou cancelamento permitidos.'
      using errcode='23503';
  end if;

  if v_sale.student_id is not null then
    select * into v_student from public.students
     where id=v_sale.student_id and owner_id=v_uid for update;
    select coalesce(sum(greatest(0,coalesce(si.aulas_contratadas,0)-coalesce(si.aulas_devolvidas,0))),0)::integer
      into v_net_lessons
      from public.sale_items si
     where si.sale_id=p_sale_id and si.owner_id=v_uid;
    if coalesce(v_student.aulas_restantes,0) < v_net_lessons then
      raise exception 'Não é possível excluir: parte das aulas desta venda já foi utilizada.';
    end if;
    update public.students
       set aulas_compradas=greatest(0,coalesce(aulas_compradas,0)-v_net_lessons),
           aulas_restantes=greatest(0,coalesce(aulas_restantes,0)-v_net_lessons)
     where id=v_sale.student_id and owner_id=v_uid;
  end if;

  delete from public.financial_transactions
   where sale_id=p_sale_id and owner_id=v_uid;
  delete from public.sales where id=p_sale_id and owner_id=v_uid;
  return jsonb_build_object('sale_id',p_sale_id,'aulas_removidas',v_net_lessons);
end;
$function$;

revoke all on function private.create_partner_commission_for_service() from public, anon, authenticated;
